move():-
****
-move is used to transfer ownership to one object to another
#include <iostream>
#include <string>
#include <utility>
using namespace std;

int main()
{
    string s1 = "Hello World";

    string s2 = move(s1);

    cout << "s2: " << s2 << endl;
    cout << "s1: " << s1 << endl;
}

swap():-
****
-it is used to exchanges the values of two variables.
#include <iostream>
#include <utility>
using namespace std;

int main()
{
    int a = 10;
    int b = 20;

    swap(a, b);

    cout << "a = " << a << endl;
    cout << "b = " << b << endl;
}
